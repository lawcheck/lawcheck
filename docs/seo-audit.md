Автопробег sitemap (все URL должны отдавать 200):

```python
import urllib.request, re, concurrent.futures
sm = urllib.request.urlopen('https://lawchek.ru/sitemap.xml', timeout=30).read().decode()
urls = re.findall(r'<loc>([^<]+)</loc>', sm)
def chk(u):
    try:
        r = urllib.request.urlopen(urllib.request.Request(u, headers={'User-Agent': 'Mozilla/5.0'}), timeout=20)
        return u, r.status
    except Exception as e:
        return u, str(e)[:40]
with concurrent.futures.ThreadPoolExecutor(10) as ex:
    for u, s in ex.map(chk, urls):
        if s != 200: print(s, u)
```
